function(slic3r_remap_configs targets from_Cfg to_Cfg)
    if(MSVC)
        string(TOUPPER ${from_Cfg} from_CFG)
        foreach(tgt ${targets})
            if(TARGET ${tgt})
                # Newer CMake/vcpkg packages frequently expose imported ALIAS
                # targets. ALIAS targets cannot be modified with
                # set_target_properties(); remap only the real target.
                get_target_property(_alias_target ${tgt} ALIASED_TARGET)
                if(NOT _alias_target)
                    set_target_properties(${tgt} PROPERTIES MAP_IMPORTED_CONFIG_${from_CFG} ${to_Cfg})
                endif()
            endif()
        endforeach()
    endif()
endfunction()

if(TARGET Boost::system)
    message(STATUS "Boost::boost exists")

    set(_boost_targets "")
    foreach(comp ${_boost_components})
        list(APPEND _boost_targets "Boost::${comp}")
    endforeach()
    target_link_libraries(boost_libs INTERFACE ${_boost_targets})
    slic3r_remap_configs("${_boost_targets}" RelWithDebInfo Release)
endif()
